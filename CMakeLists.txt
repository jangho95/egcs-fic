cmake_minimum_required(VERSION 3.16)
project(egcs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
find_package(Threads REQUIRED)
add_library(egcs_core STATIC
    src/sim.cpp src/path.cpp src/cost.cpp src/dispatchers.cpp
    src/ga2022.cpp src/trace.cpp)
target_include_directories(egcs_core PUBLIC src)
target_link_libraries(egcs_core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(egcs_core PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
function(egcs_program name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE egcs_core)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -O2 -Wall -Wextra -Wpedantic)
    endif()
endfunction()
egcs_program(egcs src/main.cpp)
egcs_program(confirmation research/confirmation30_main.cpp)
file(GLOB EGCS_RESEARCH CONFIGURE_DEPENDS research/*.cpp)
foreach(source IN LISTS EGCS_RESEARCH)
    get_filename_component(name ${source} NAME_WE)
    if(NOT name STREQUAL "confirmation30_main")
        egcs_program(${name} ${source})
    endif()
endforeach()
file(GLOB EGCS_TESTS CONFIGURE_DEPENDS tests/*.cpp)
foreach(source IN LISTS EGCS_TESTS)
    get_filename_component(name ${source} NAME_WE)
    egcs_program(${name} ${source})
endforeach()
enable_testing()
add_test(NAME simulator COMMAND egcs selftest)
add_test(NAME pickup_conventions COMMAND pickup_convention_check)
add_test(NAME dqn_initial COMMAND dqn_check)
add_test(NAME dqn_followup COMMAND dqn2_check)
add_test(NAME dqn_final COMMAND dqn3_check)
add_test(NAME excluded_counter COMMAND excluded_count --selftest)
add_test(NAME observation_projection COMMAND observation_projection)
add_test(NAME insertion COMMAND sequential_greedy_probe)
add_test(NAME forward_boundaries COMMAND forward_dispatch_boundaries)
add_test(NAME combined_boundaries COMMAND structured_dispatch_boundaries)
add_test(NAME closed_loop_boundary COMMAND structured_policy_engine --boundary)
